    ```ts
    import { describe, expect, it, vi } from 'vitest'
    import { handleFill } from '../src/runtime/command-handlers'
    import type { CommandHandlerDeps } from '../src/runtime/command-handlers'
    import type { EventSequences } from '../src/runtime/event-sequences'

    function makeStubEventSequences(): EventSequences {
      return {
        click: vi.fn().mockResolvedValue(undefined),
        dblclick: vi.fn().mockResolvedValue(undefined),
        contextmenu: vi.fn().mockResolvedValue(undefined),
        hover: vi.fn().mockResolvedValue(undefined),
        longpress: vi.fn().mockResolvedValue(undefined),
        mousePressed: vi.fn().mockResolvedValue(undefined),
        mouseMoved: vi.fn().mockResolvedValue(undefined),
        mouseReleased: vi.fn().mockResolvedValue(undefined),
        pointerDrag: vi.fn().mockResolvedValue(undefined),
        wheel: vi.fn().mockResolvedValue(undefined),
        htmlDrag: vi.fn().mockResolvedValue(undefined),
        insertText: vi.fn().mockResolvedValue(undefined),
        typeText: vi.fn().mockResolvedValue(undefined),
        pressKey: vi.fn().mockResolvedValue(undefined),
        selectAllAndDelete: vi.fn().mockResolvedValue(undefined),
      } as unknown as EventSequences
    }

    function makeDepsFor(element: HTMLElement, descriptor: any): CommandHandlerDeps {
      const emptySnapshot = { version: 1, targets: [], flowOverlay: null } as any
      return {
        captureSnapshot: () => emptySnapshot,
        captureSettledSnapshot: async () => emptySnapshot,
        getDescriptors: () => [descriptor],
        resolveExecutionConfig: (p) => ({
          pointerAnimation: false,
          clickDelayMs: 0,
          pointerDurationMs: 0,
          autoScroll: true,
          auroraGlow: false,
          auroraTheme: 'aurora-night',
          cursorName: 'default',
          ...(p ?? {}),
        }) as any,
        queue: { active: false, push: async (t: any) => { await t.execute?.() } } as any,
        eventSequences: makeStubEventSequences(),
      } as CommandHandlerDeps
    }

    function makeDescriptor(element: HTMLElement, targetId: string) {
      return {
        actionKinds: ['fill'] as const,
        groupId: 'g-1',
        target: { targetId, selector: `#${element.id}` },
      }
    }

    describe('handleFill — CDP Input domain', () => {
      it('uses insertText for a plain text input (strategy=auto)', async () => {
        document.body.innerHTML = '<input id="x" type="text" />'
        const el = document.getElementById('x') as HTMLInputElement
        Object.defineProperty(el, 'getBoundingClientRect', {
          value: () => ({ x: 0, y: 0, top: 0, left: 0, right: 50, bottom: 20, width: 50, height: 20 }),
        })
        const deps = makeDepsFor(el, makeDescriptor(el, 'x'))
        const result = await handleFill(deps, { targetId: 'x', value: 'hello', clear: true })
        expect((deps.eventSequences.insertText as any)).toHaveBeenCalledWith('hello')
        expect((deps.eventSequences.selectAllAndDelete as any)).toHaveBeenCalled()
      })

      it('uses typeText for a masked tel input', async () => {
        document.body.innerHTML = '<input id="p" type="tel" pattern="\\\\d{3}-\\\\d{4}" inputmode="tel" />'
        const el = document.getElementById('p') as HTMLInputElement
        Object.defineProperty(el, 'getBoundingClientRect', {
          value: () => ({ x: 0, y: 0, top: 0, left: 0, right: 50, bottom: 20, width: 50, height: 20 }),
        })
        const deps = makeDepsFor(el, makeDescriptor(el, 'p'))
        await handleFill(deps, { targetId: 'p', value: '010-1234', strategy: 'auto', clear: true })
        expect((deps.eventSequences.typeText as any)).toHaveBeenCalled()
        expect((deps.eventSequences.insertText as any)).not.toHaveBeenCalled()
      })

      it('handles contenteditable with insertText', async () => {
        document.body.innerHTML = '<div id="ce" contenteditable="true"></div>'
        const el = document.getElementById('ce') as HTMLElement
        Object.defineProperty(el, 'getBoundingClientRect', {
          value: () => ({ x: 0, y: 0, top: 0, left: 0, right: 100, bottom: 20, width: 100, height: 20 }),
        })
        Object.defineProperty(el, 'isContentEditable', { value: true, configurable: true })
        const deps = makeDepsFor(el, makeDescriptor(el, 'ce'))
        await handleFill(deps, { targetId: 'ce', value: 'hi', clear: false })
        expect((deps.eventSequences.insertText as any)).toHaveBeenCalledWith('hi')
      })

      it('does not call selectAllAndDelete when clear=false', async () => {
        document.body.innerHTML = '<input id="y" type="text" value="existing" />'
        const el = document.getElementById('y') as HTMLInputElement
        Object.defineProperty(el, 'getBoundingClientRect', {
          value: () => ({ x: 0, y: 0, top: 0, left: 0, right: 50, bottom: 20, width: 50, height: 20 }),
        })
        const deps = makeDepsFor(el, makeDescriptor(el, 'y'))
        await handleFill(deps, { targetId: 'y', value: 'more', clear: false })
        expect((deps.eventSequences.selectAllAndDelete as any)).not.toHaveBeenCalled()
      })
    })
    ```

    **주의:** `handleFill`이 통과하는 가드(visibility, viewport, topmost, enabled)를 만족시키기 위해 테스트에서 `getBoundingClientRect`·`isVisible` 등을 stub해야 한다. 실제 구현 시 기존 `runtime.spec.ts`의 도우미 패턴(예: `setupDocument()`)이 있으면 재사용하고, 없으면 테스트에서 `vi.mock`으로 가드 함수를 우회(예: `vi.mock('../src/runtime/dom-utils', async (orig) => { const a = await orig() as any; return { ...a, isVisible: () => true, isElementInViewport: () => true, isTopmostInteractable: () => true, isEnabled: () => true } })`).
