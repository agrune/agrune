    **2. `src/fiber/identity-index.ts` — FiberIdentityIndex (값 타입 저장, 참조 아님):**

    RESEARCH §"FiberIdentityIndex 전체 구조" 코드 블록을 **그대로 채택**. 핵심:

    ```typescript
    import {
      isHostFiber,
      isCompositeFiber,
      getDisplayName,
      getFiberStack,
      getLatestFiber,
      type Fiber,
    } from 'bippy'
    import type { FiberIdentityPath, FiberPathSegment } from '@agrune/manifest'

    export class FiberIdentityIndex {
      private readonly domToPath = new WeakMap<HTMLElement, FiberIdentityPath>()
      private readonly pathToRef = new Map<string, WeakRef<HTMLElement>>()

      indexFiber(fiber: Fiber): void { /* isHostFiber guard → path 생성 → set */ }
      deindexFiber(fiber: Fiber): void { /* cleanup */ }
      getByPath(path: FiberIdentityPath): HTMLElement | null { /* deref or null */ }
      private buildPath(stack: Fiber[]): FiberIdentityPath {
        return stack
          .filter(isCompositeFiber)
          .slice(0, 8)  // RESEARCH: depth cap (DoS 방어)
          .map(f => ({
            componentName: getDisplayName(f.type) ?? '',
            key: f.key as string | null,  // bippy Fiber.key typing
            index: f.index,
          }))
      }
    }
    ```

    **중요 설계 결정:**
    - `domToPath` 를 `WeakMap<HTMLElement, FiberIdentityPath>` 로 저장 → DOM GC 시 자동 제거.
    - `pathToRef` 는 `Map<string, WeakRef<HTMLElement>>` — `JSON.stringify(path)` 를 key로 사용. `WeakRef.deref()` 로 element 살아있는지 확인.
    - **fiber object 자체를 저장하지 말 것** (RESEARCH Anti-Patterns) — double-buffering + GC 방해.
    - `slice(0, 8)` 로 path depth 8 cap — T-13-04 DoS 방어.
    - `buildPath` 호출 전 반드시 `getLatestFiber(fiber)` 로 alternate 해결.

    **5. `src/guard/ssr-barrier.ts` — hydration barrier:**

    RESEARCH Pattern 6 그대로:
    ```typescript
    import { _fiberRoots } from 'bippy'

    export function waitForHydration(): Promise<void> {
      return new Promise((resolve) => {
        if (typeof document === 'undefined') {
          // SSR: never resolves (useEffect 자체가 client에서만 호출되므로 reach 불가)
          return
        }
        const ready = () => document.readyState === 'complete' && _fiberRoots.size > 0
        if (ready()) {
          resolve()
          return
        }
        const check = () => {
          if (ready()) resolve()
        }
        window.addEventListener('DOMContentLoaded', check)
        window.addEventListener('load', check)
        // 마지막 fallback: load 후 1 tick 대기 (RESEARCH Pattern 6)
        window.addEventListener('load', () => setTimeout(() => resolve(), 0), { once: true })
      })
    }
    ```

      instrument,
      secure,
      traverseRenderedFibers,
      getLatestFiber,
      isHostFiber,
    } from 'bippy'
    import { FiberIdentityIndex } from '../fiber/identity-index.js'
    import { activateBridge } from '../bridge/identity-bridge.js'
    import { isProdEnabled } from '../guard/prod-guard.js'
    import { waitForHydration } from '../guard/ssr-barrier.js'

    export interface AgruneDevtoolsProps {
      manifest: AgruneManifest
      mode: 'dev' | 'prod'
    }

    export function AgruneDevtools({ manifest, mode }: AgruneDevtoolsProps): ReactElement | null {
      const activatedRef = useRef(false)

      useEffect(() => {
        if (activatedRef.current) return
        if (!isProdEnabled(mode)) return  // Guard 1+2
        if (typeof window === 'undefined') return  // SSR path (useEffect don't run, but defensive)

        let cancelled = false
        waitForHydration().then(() => {
          if (cancelled) return

          const index = new FiberIdentityIndex()

          instrument(
            secure(
              {
                onCommitFiberRoot(_rendererID, root) {
                  traverseRenderedFibers(root, (fiber, phase) => {
                    if (phase === 'unmount') {
                      if (isHostFiber(fiber)) index.deindexFiber(fiber)
                      return
                    }
                    if (!isHostFiber(fiber)) return
                    const latest = getLatestFiber(fiber)
                    index.indexFiber(latest)
                  })
                },
              },
              // Pitfall 7: mode='prod' 에서 bippy 자체 production guard를 우회해야 함
              // — Phase 13의 guard가 이미 통과했으므로.
              mode === 'prod' ? { dangerouslyRunInProduction: true } : undefined,
            ),
          )

          activateBridge(index)
          activatedRef.current = true
        })

        return () => { cancelled = true }
      }, [mode])  // manifest는 useEffect 내부에서 현재 값 사용 — re-mount 불필요

      // 이 컴포넌트는 시각적 출력 없음 (bridge 설치 전용).
      // manifest 값은 currently consume 없음 (Phase 14+ 에서 MacroRunner 연계 시 사용 예정).
      return null
    }
