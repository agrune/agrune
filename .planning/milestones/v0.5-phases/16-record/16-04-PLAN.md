    ```typescript
    import { describe, it, expect } from 'vitest'
    import { JSDOM } from 'jsdom'  // vitest 이미 jsdom env, 이걸로 HTML 파싱
    import { isSensitive } from '../src/runtime/dom-utils.js'
    import { loginFixtures } from './fixtures/corpus/login.js'
    import { paymentFixtures } from './fixtures/corpus/payment.js'
    import { signupFixtures } from './fixtures/corpus/signup.js'
    import { profileFixtures } from './fixtures/corpus/profile.js'

    const ALL_FIXTURES = [...loginFixtures, ...paymentFixtures, ...signupFixtures, ...profileFixtures]

    const PRECISION_THRESHOLD = 0.90
    const RECALL_THRESHOLD = 0.95

    function evaluate() {
      let tp = 0, fp = 0, fn = 0, tn = 0
      for (const fx of ALL_FIXTURES) {
        document.body.innerHTML = fx.html
        for (let i = 0; i < fx.elements.length; i++) {
          const el = document.querySelector(fx.elements[i]) as HTMLElement | null
          if (!el) throw new Error(`fixture ${fx.id}: selector not found ${fx.elements[i]}`)
          const predicted = isSensitive(el)
          const actual = fx.expected[i]
          if (predicted && actual) tp++
          else if (predicted && !actual) fp++
          else if (!predicted && actual) fn++
          else tn++
        }
      }
      const precision = tp / (tp + fp || 1)
      const recall = tp / (tp + fn || 1)
      return { tp, fp, fn, tn, precision, recall, total: ALL_FIXTURES.length }
    }

    describe('Phase 16 RECORD-04: sensitive corpus precision/recall', () => {
      it('corpus size ≥ 100', () => {
        expect(ALL_FIXTURES.length).toBeGreaterThanOrEqual(100)
      })

      it('precision ≥ 0.90, recall ≥ 0.95', () => {
        const m = evaluate()
        console.log(`corpus=${m.total} tp=${m.tp} fp=${m.fp} fn=${m.fn} tn=${m.tn}`)
        console.log(`precision=${m.precision.toFixed(3)} recall=${m.recall.toFixed(3)}`)
        expect(m.precision).toBeGreaterThanOrEqual(PRECISION_THRESHOLD)
        expect(m.recall).toBeGreaterThanOrEqual(RECALL_THRESHOLD)
      })

      // 카테고리별 세부 보고 (CI 디버깅용, threshold 없음)
      it('reports per-category metrics', () => {
        for (const category of ['login', 'payment', 'signup', 'profile'] as const) {
          const subset = ALL_FIXTURES.filter(f => f.category === category)
          // 집계 후 console.log — 실패 시 어느 카테고리 문제인지 파악
          expect(subset.length).toBeGreaterThan(0)
        }
      })
    })
    ```

    # Manifest Authoring Skill

    ## When to use
    - React 프로젝트에 agrune runtime 을 통합할 때
    - 외부 사이트용 manifest 를 작성할 때 (CSS selector fallback 전용)

    ## Core workflow

    1. **프로젝트 분석**: interactive element (button, input, select, a, [role=*]) 탐색
    2. **AgruneDevtools 마운트 확인**: root-import 1줄 추가 여부 확인 (React 전용)

       import { AgruneDevtools } from '@agrune/react'
       <AgruneDevtools manifest={m} mode="dev" />
       \`\`\`
    3. **manifest.ts 생성**: `defineManifest({ version: 3, targets: [...] })`
    4. **Target 정의 규칙**:
       - `targetId`: snake_case, lowercase, fiber componentName 기반
       - `selector`: `{ fiber?: { path }, role?, css? }` — fiber path 는 React 프로젝트만
       - `actionKinds`: `['click']` | `['fill']` | `['click','fill']`

    5. **validate**: `agrune manifest validate src/manifest.ts`
