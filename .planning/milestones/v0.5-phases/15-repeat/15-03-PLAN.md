    ```typescript
    const REPEATED_TARGET_KEY_DELIMITER = '__agrune_repeatKey_'  // runtime과 동일 상수
    // D-06: 필요시 @agrune/core에서 재export 하도록 별도 commit 가능 — 현재는 MCP-local 상수

    export class AgentTargetIdParseError extends Error {
      constructor(
        public readonly input: string,
        message: string,
      ) {
        super(message)
        this.name = 'AgentTargetIdParseError'
      }
    }

    /**
     * AI-facing dot-bracket targetId → runtime delimiter 형식으로 정규화.
     * Accepts:
     *   - `plain_id` → passthrough
     *   - `repeatId[key=value].baseTargetId` → `repeatId__agrune_repeatKey_value.baseTargetId`
     *   - `repeatId__agrune_repeatKey_value.baseTargetId` → passthrough (이미 normalized)
     *
     * Rejects (throws AgentTargetIdParseError):
     *   - bracket 불균형
     *   - bracket 내부에 `=` 없음
     *   - 빈 key
     *   - bracket 후 `.` 없음
     *
     * Linear-time: 단일 pass scan, regex backtracking 없음.
     */
    export function normalizeAgentTargetId(input: string): string {
      // early-return: bracket 미포함 → passthrough
      const bracketStart = input.indexOf('[')
      if (bracketStart < 0) return input
      if (bracketStart === 0) {
        // 시작이 bracket이면 invalid (repeatId 필요)
        throw new AgentTargetIdParseError(input, 'Missing repeatId before bracket')
      }
      // 이미 runtime 형식이면 passthrough (bracket 위치가 delimiter 뒤가 아닐 때)
      if (input.includes(REPEATED_TARGET_KEY_DELIMITER)) return input

      const repeatId = input.slice(0, bracketStart)
      // rightmost `]` for nested bracket support (Test 4)
      const bracketEnd = input.lastIndexOf(']')
      if (bracketEnd <= bracketStart) {
        throw new AgentTargetIdParseError(input, 'Unbalanced brackets')
      }
      const bracketInner = input.slice(bracketStart + 1, bracketEnd)
      const eqIdx = bracketInner.indexOf('=')
      if (eqIdx < 0) {
        throw new AgentTargetIdParseError(input, 'Bracket must contain "=" (keyFromField=value)')
      }
      // keyFromField는 validation 용도로만 읽고 normalized output에는 포함하지 않음 — 실제 키 값만 사용
      const value = bracketInner.slice(eqIdx + 1).trim()
      if (!value) {
        throw new AgentTargetIdParseError(input, 'Bracket value cannot be empty')
      }
      const afterBracket = input.slice(bracketEnd + 1)
      if (!afterBracket.startsWith('.')) {
        throw new AgentTargetIdParseError(input, 'Expected "." after "]"')
      }
      const baseTargetId = afterBracket.slice(1)
      if (!baseTargetId) {
        throw new AgentTargetIdParseError(input, 'baseTargetId cannot be empty')
      }
      // ReDoS 내성: 위 로직은 모두 indexOf/slice (O(n)), regex 없음
      return `${repeatId}${REPEATED_TARGET_KEY_DELIMITER}${value}.${baseTargetId}`
    }
    ```
