    ```typescript
    import type { ManifestRepeat, SelectorLadder } from '../types'
    import { isElementInViewport } from './dom-utils'
    import { resolveByLadder } from './target-resolver'

    export const REPEAT_MAX_INSTANCES = 1000

    export interface RepeatInstance {
      el: HTMLElement
      key: string
      index: number
    }

    export interface VirtualizedExpandResult {
      instances: RepeatInstance[]
      logicalSize: number | null
    }

    export class RepeatExpander {
      /** DOM strategy: container 내부 전체 row 열거. */
      expand(repeat: ManifestRepeat, container?: HTMLElement): RepeatInstance[] {
        const scope = container ?? document
        const rowCss = this.resolveRowSelectorCss(repeat)
        const all = Array.from(scope.querySelectorAll<HTMLElement>(rowCss))
        const capped = this.applyCap(all, repeat.repeatId)
        return this.buildInstances(capped, repeat.keyFrom, repeat.repeatId)
      }

      /** Virtualized strategy: viewport 내부 row만 + aria-rowcount READ-ONLY. */
      expandVirtualized(
        repeat: ManifestRepeat,
        container?: HTMLElement,
      ): VirtualizedExpandResult {
        const scope = container ?? document
        const rowCss = this.resolveRowSelectorCss(repeat)
        const all = Array.from(scope.querySelectorAll<HTMLElement>(rowCss))
        const visible = all.filter((el) => isElementInViewport(el))
        const capped = this.applyCap(visible, repeat.repeatId)
        const instances = this.buildInstances(capped, repeat.keyFrom, repeat.repeatId)
        const logicalSize = this.readLogicalSize(container ?? null)
        return { instances, logicalSize }
      }

      // --- helpers ---

      private resolveRowSelectorCss(repeat: ManifestRepeat): string {
        // repeat.targets[0].selector 를 row element selector로 사용.
        // Decision A: containerSelector가 있으면 그 안에서 targets[0]을 row selector로 사용.
        // 각 row 내부에 여러 targets (actionable element들)가 있을 수 있으나,
        // row 자체를 찾기 위해 template/첫 target의 css를 row-level로 간주.
        // keyFrom이 `el.dataset.X`처럼 row element에 접근 — row selector는 targets[0] 기반.
        const firstTarget = repeat.targets[0]
        if (!firstTarget) return '*'  // defensive
        const ladder = firstTarget.selector
        return ladder.css ?? ladder.attr ?? this.ladderToCss(ladder)
      }

      private ladderToCss(ladder: SelectorLadder): string {
        if (ladder.testId) return `[data-testid="${CSS.escape(ladder.testId)}"]`
        if (ladder.role) return `[role="${CSS.escape(ladder.role.name)}"]`
        return '*'  // worst fallback
      }

      private applyCap<T>(list: T[], repeatId: string): T[] {
        if (list.length <= REPEAT_MAX_INSTANCES) return list
        console.warn(
          `[agrune] RepeatExpander: repeat "${repeatId}" truncated from ${list.length} to ${REPEAT_MAX_INSTANCES} instances (DoS cap)`,
        )
        return list.slice(0, REPEAT_MAX_INSTANCES)
      }

      private buildInstances(
        els: HTMLElement[],
        keyFromExpr: string,
        repeatId: string,
      ): RepeatInstance[] {
        let keyFn: ((el: HTMLElement) => string) | null = null
        try {
          // T-14-07 선례: new Function 스코프 격리 (eval() 금지)
          keyFn = new Function('el', `return String(${keyFromExpr})`) as (el: HTMLElement) => string
        } catch (err) {
          // CSP `unsafe-eval` 차단 또는 문법 오류 (Pitfall 2)
          console.warn(`[agrune] RepeatExpander: keyFrom compile failed for repeat "${repeatId}":`, err)
        }

        const seen = new Map<string, number>()
        return els.map((el, index) => {
          let key: string
          if (keyFn) {
            try {
              const v = keyFn(el)
              key = (v === undefined || v === null || v === 'undefined' || v === 'null') ? `__idx_${index}` : String(v)
            } catch (err) {
              console.warn(`[agrune] RepeatExpander: keyFrom eval failed for repeat "${repeatId}" at index ${index}:`, err)
              key = `__idx_${index}`
            }
          } else {
            key = `__idx_${index}`
          }
          // Pitfall 3: 중복 키 처리
          if (seen.has(key)) {
            const suffix = `${key}__dup_${index}`
            console.warn(`[agrune] RepeatExpander: duplicate key "${key}" for repeat "${repeatId}", using "${suffix}"`)
            seen.set(suffix, index)
            return { el, key: suffix, index }
          }
          seen.set(key, index)
          return { el, key, index }
        })
      }

      private readLogicalSize(container: HTMLElement | null): number | null {
        if (!container) return null
        // Pitfall 5: READ-ONLY (inject 금지)
        const rowCount = container.getAttribute('aria-rowcount')
        if (rowCount) {
          const n = Number.parseInt(rowCount, 10)
          if (Number.isFinite(n) && n >= 0) return n
        }
        const setSize = container.getAttribute('aria-setsize')
        if (setSize) {
          const n = Number.parseInt(setSize, 10)
          if (Number.isFinite(n) && n >= 0) return n
        }
        return null
      }
    }
    ```
