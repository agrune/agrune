    ```typescript
    export function collectDescriptors(manifest: AgruneManifest): TargetDescriptor[] {
      const result: TargetDescriptor[] = []
      for (const group of manifest.groups) {
        for (const target of group.targets) {
          const kinds = target.actionKinds.filter((k) => VALID_ACTIONS.has(k))
          if (kinds.length === 0) continue
          result.push({
            actionKinds: [...new Set(kinds)] as ActionKind[],
            groupId: group.groupId,
            groupName: group.name,
            groupDesc: group.desc,
            target,
          })
        }
        for (const repeat of group.repeats ?? []) {
          for (const target of repeat.targets) {
            const kinds = target.actionKinds.filter((k) => VALID_ACTIONS.has(k))
            if (kinds.length === 0) continue
            result.push({
              actionKinds: [...new Set(kinds)] as ActionKind[],
              groupId: group.groupId,
              groupName: group.name,
              groupDesc: group.desc,
              target,
            })
          }
        }
      }
      return result.sort((a, b) => a.target.targetId.localeCompare(b.target.targetId))
    }

          rawAction.split(',').map(a => a.trim()).filter(a => VALID_ACTIONS.has(a as ActionKind))
        )] as ActionKind[]
        if (kinds.length === 0) return

        const key = element.getAttribute('data-agrune-key')?.trim()
        const groupEl = element.closest<HTMLElement>(LIVE_SCAN_GROUP_SELECTOR)
        const groupId = groupEl?.getAttribute('data-agrune-group')?.trim() || LIVE_SCAN_DEFAULT_GROUP_ID

        result.push({
          actionKinds: kinds,
          groupId,
          groupName: groupEl?.getAttribute('data-agrune-group-name') || (
            groupId === LIVE_SCAN_DEFAULT_GROUP_ID ? LIVE_SCAN_DEFAULT_GROUP_NAME : groupId
          ),
          groupDesc: groupEl?.getAttribute('data-agrune-group-desc') || undefined,
          target: {
            targetId: key || `agrune_${index}`,
            name: element.getAttribute('data-agrune-name') ?? undefined,
            desc: element.getAttribute('data-agrune-desc') ?? undefined,
            actionKinds: kinds,
            selector: { css: buildLiveSelector(element) } as SelectorLadder,
          },
        })
      })

      return result.sort((a, b) => a.target.targetId.localeCompare(b.target.targetId))
    }
    ```

    `mergeDescriptors` 는 `selector: string` 필드 참조를 `selector: SelectorLadder` 참조로 수정. `existing.target.selector` 를 그대로 복사 (객체).

    `captureTarget` 내 `isSensitive` 호출을 manifest flag 를 포함하도록 업데이트:
    ```typescript
    // captureTarget 내부
    const sensitive = isSensitive(element, (descriptor.target.sensitive === true ? true : undefined))
    ```

    `PageTarget.selector` 직렬화 — v3 shape 이 object 이므로 snapshot 직렬화 시 string 표현이 필요한 기존 consumer 가 있다면 `JSON.stringify(selector)` 로 대체. `PageTarget.selector` 는 `@agrune/core` 에서 현재 `string` 타입이므로, snapshot.ts 에서 `captureTarget` 출력 시 `selector: JSON.stringify(descriptor.target.selector)` 로 임시 serialize (Plan 12 에서 PageSnapshot v3 shape 으로 교체 예정).

      scannedTargets: ScannedTarget[],
      scannedGroups: ScannedGroup[],
    ): AgruneManifest {
      const groupMap = new Map<string, ManifestGroup>()
      for (const g of scannedGroups) {
        groupMap.set(g.groupId, {
          groupId: g.groupId,
          name: g.name ?? undefined,
          desc: g.desc ?? undefined,
          targets: [],
        })
      }
      for (const t of scannedTargets) {
        const group = groupMap.get(t.groupId) ?? {
          groupId: t.groupId,
          targets: [],
        }
        groupMap.set(t.groupId, group)
        group.targets.push({
          targetId: t.targetId,
          name: t.name ?? undefined,
          desc: t.desc ?? undefined,
          actionKinds: t.actions as ActionKind[],
          selector: { css: t.selector } as SelectorLadder,
        })
      }
      return {
        version: 3,
        groups: Array.from(groupMap.values()),
      }
    }
