    2. `registry-seed/.github/scripts/pr-bot.mjs`:
       ```javascript
       import { Octokit } from '@octokit/rest'
       import * as core from '@actions/core'
       import { execSync } from 'node:child_process'
       import { readFileSync } from 'node:fs'

       const octokit = new Octokit({ auth: process.env.GITHUB_TOKEN })
       const [owner, repo] = process.env.GITHUB_REPOSITORY.split('/')
       const prNumber = Number(process.env.PR_NUMBER)
       const prAuthor = process.env.PR_AUTHOR

       const labels = new Set()

       // 1. fetch changed files (manifests/**)
       const { data: files } = await octokit.pulls.listFiles({ owner, repo, pull_number: prNumber })
       const manifestFiles = files.filter(f => f.filename.startsWith('manifests/') && f.filename.endsWith('.json'))

       // 2. for each file: compare before / after, detect sensitive transitions
       function collectSensitiveMap(entry) {
         const map = new Map()
         for (const g of entry?.manifest?.groups ?? []) {
           for (const t of g.targets ?? []) {
             if (t.sensitive === true) map.set(t.targetId, true)
           }
         }
         return map
       }
       for (const f of manifestFiles) {
         let before = null
         try {
           const raw = execSync(`git show origin/main:${f.filename}`, { encoding: 'utf-8' })
           before = JSON.parse(raw)
         } catch { /* new file */ }
         const after = JSON.parse(readFileSync(f.filename, 'utf-8'))

         const beforeSensitive = collectSensitiveMap(before)
         const afterSensitive = collectSensitiveMap(after)

         // Pitfall 8: sensitive:true 필드 삭제 또는 false 전환 감지
         for (const [tid, was] of beforeSensitive) {
           if (was && !afterSensitive.get(tid)) {
             labels.add('requires-human-review:sensitive')
             core.notice(`[${f.filename}] sensitive:true → removed/false on ${tid}`, { file: f.filename })
           }
         }

         // Pitfall 7: community tier + prod 요청 → schema-fail (이건 schema level 에서도 거부되지만 이중 확인)
         if (after?.registry?.tier === 'community' && after?.registry?.allowedEnvironments?.includes('prod')) {
           labels.add('schema-fail')
         }
         if (before?.registry?.tier === 'community' && after?.registry?.tier === 'verified') {
           labels.add('tier-escalation')
         }
       }

       // 3. Velocity check: PR author 의 최근 merged PR 수
       const { data: authorPrs } = await octokit.search.issuesAndPullRequests({
         q: `repo:${owner}/${repo} author:${prAuthor} is:pr is:merged`,
       })
       const maintainers = JSON.parse(readFileSync('maintainers.json', 'utf-8')).maintainers
       const isMaintainer = maintainers.includes(prAuthor)
       if (!isMaintainer && authorPrs.total_count < 3) {
         // 최근 PR 중 30일 이내 것 있으면 holddown
         const cutoff = Date.now() - 30 * 24 * 60 * 60 * 1000
         const recent = authorPrs.items.filter(p => new Date(p.closed_at ?? 0).getTime() > cutoff)
         if (recent.length > 0) labels.add('velocity:holddown')
       }

       // 4. DNS check for seedUrl — Pitfall 4 (seed URL privacy leak)
       for (const f of manifestFiles) {
         const entry = JSON.parse(readFileSync(f.filename, 'utf-8'))
         const url = entry?.registry?.seedUrl
         if (url) {
           const parsed = new URL(url)
           if (
             parsed.hostname === 'localhost'
             || parsed.hostname.endsWith('.internal')
             || /^10\.|^192\.168\.|^172\.(1[6-9]|2\d|3[01])\./.test(parsed.hostname)
             || parsed.protocol !== 'https:'
           ) {
             labels.add('schema-fail')
             core.notice(`[${f.filename}] seedUrl private or non-https: ${url}`)
           }
         }
       }

       // 5. apply labels
       if (labels.size > 0) {
         await octokit.issues.addLabels({ owner, repo, issue_number: prNumber, labels: [...labels] })
       }
       core.setOutput('labels', [...labels].join(','))
       ```

    3. `registry-seed/.github/scripts/health-check.mjs`:
       ```javascript
       import { chromium } from 'playwright'
       import { Octokit } from '@octokit/rest'
       import * as core from '@actions/core'
       import { readdirSync, readFileSync, writeFileSync } from 'node:fs'

       const STATE_FILE = '.github/health-state.json'  // 2주 연속 fail 추적
       let state = {}
       try { state = JSON.parse(readFileSync(STATE_FILE, 'utf-8')) } catch {}

       const browser = await chromium.launch({ headless: true })
       const context = await browser.newContext({ userAgent: 'agrune-health-check/0.5' })
       const page = await context.newPage()

       const files = readdirSync('manifests').filter(n => n.endsWith('.json'))
       const fails = []

       for (const name of files) {
         const entry = JSON.parse(readFileSync(`manifests/${name}`, 'utf-8'))
         const seedUrl = entry?.registry?.seedUrl
         if (!seedUrl) continue

         let resolved = 0, attempted = 0
         try {
           await page.goto(seedUrl, { waitUntil: 'domcontentloaded', timeout: 15000 })
           const targets = (entry.manifest?.groups ?? []).flatMap(g => g.targets ?? [])
           const sample = targets.slice(0, 3)  // Pitfall: 외부 사이트 부하 최소화
           for (const t of sample) {
             attempted++
             const [firstLadder] = t.selector
             // snapshot only — NO click/fill (RESEARCH Anti-Pattern: Health check 에서 실제 action 금지)
             let matched = false
             if (firstLadder.type === 'role' && firstLadder.role) {
               const loc = page.getByRole(firstLadder.role, firstLadder.nameRegex ? { name: new RegExp(firstLadder.nameRegex) } : undefined)
               if ((await loc.count()) > 0) matched = true
             } else if (firstLadder.type === 'text' && firstLadder.text) {
               if ((await page.getByText(firstLadder.text).count()) > 0) matched = true
             } else if (firstLadder.type === 'css' && firstLadder.css) {
               if ((await page.locator(firstLadder.css).count()) > 0) matched = true
             }
             if (matched) resolved++
           }
         } catch (err) {
           core.notice(`[${name}] navigation failed: ${err.message}`)
         }

         const key = name
         const ok = resolved > 0
         state[key] ??= { consecutiveFails: 0, lastCheckedAt: null }
         state[key].lastCheckedAt = new Date().toISOString()
         if (ok) {
           state[key].consecutiveFails = 0
         } else {
           state[key].consecutiveFails = (state[key].consecutiveFails ?? 0) + 1
           fails.push({ name, consecutive: state[key].consecutiveFails })
         }
       }
       writeFileSync(STATE_FILE, JSON.stringify(state, null, 2) + '\n')
       await browser.close()
       ```
