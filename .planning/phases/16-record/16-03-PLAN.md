    ```typescript
    import { Project, SyntaxKind, IndentationText, ObjectLiteralExpression } from 'ts-morph'
    import { createTwoFilesPatch } from 'diff'  // 이미 npm 에 있음, 필요시 수동 라인 diff 구현
    import { validateManifest } from '@agrune/manifest'
    import { PendingStore } from './pending-store.js'
    import type { PendingCaptureFile } from './pending-store.js'

    export class MergeError extends Error {
      constructor(public readonly code: string, message: string) { super(message) }
    }

    export interface MergeResult { merged: string; diff: string; addedTargetIds: string[] }

    export function mergeTargetIntoManifest(
      sourceText: string,
      pending: PendingCaptureFile,
      manifestFilePath: string,
    ): MergeResult {
      // 1. Pending sanitize (T-16-09)
      for (const t of pending.targets) {
        PendingStore.sanitizeTargetId(t.targetId)
      }

      // 2. Selector validate (hash class / nth-child 금지 재확인)
      // validateManifest 는 full manifest 를 기대하므로 임시 manifest 만들어 통과 여부만 확인
      // 또는 validator.ts 의 assertNoHashClass / assertNoNthChild 를 직접 호출
      // (11-01 SUMMARY 에서 export 확인됨)

      // 3. ts-morph Project 설정 — trailing comma 감지
      const hasTrailingComma = /,\s*[\]\)\}]/.test(sourceText)
      const project = new Project({
        useInMemoryFileSystem: false,  // 실제 파일 읽지 않고 text 기반
        skipAddingFilesFromTsConfig: true,
        manipulationSettings: {
          useTrailingCommas: hasTrailingComma,
          indentationText: /\n\t/.test(sourceText) ? IndentationText.Tab : IndentationText.TwoSpaces,
        },
      })
      const sf = project.createSourceFile(manifestFilePath, sourceText, { overwrite: true })

      // 4. defineManifest call 찾기
      const call = sf.getDescendantsOfKind(SyntaxKind.CallExpression)
        .find(c => c.getExpression().getText() === 'defineManifest')
      if (!call) throw new MergeError('CALL_NOT_FOUND', 'defineManifest call not found')
      const arg = call.getArguments()[0]
      if (!arg || arg.getKind() !== SyntaxKind.ObjectLiteralExpression) {
        throw new MergeError('INLINE_REQUIRED', 'defineManifest argument must be inline object literal')
      }
      const configObj = arg as ObjectLiteralExpression

      // 5. targets 또는 groups[0].targets 찾기
      const targetsProp = configObj.getProperty('targets')
      const groupsProp = configObj.getProperty('groups')
      const arr = findOrCreateTargetsArray(configObj, targetsProp, groupsProp)

      // 6. 기존 targetId 수집 (중복 검사, T-16-04 ~)
      const existingIds = collectExistingTargetIds(arr)
      const added: string[] = []
      for (const t of pending.targets) {
        if (existingIds.has(t.targetId)) {
          throw new MergeError('DUPLICATE_TARGET', `duplicate targetId: ${t.targetId}`)
        }
        const elementText = buildDefineTargetText(t)
        arr.addElement(elementText)
        added.push(t.targetId)
      }

      // 7. 결과 추출 + diff
      const merged = sf.getFullText()
      const diff = createTwoFilesPatch(manifestFilePath, manifestFilePath, sourceText, merged)
      return { merged, diff, addedTargetIds: added }
    }

    function buildDefineTargetText(target: { targetId: string; selector: SelectorLadder; sensitive?: true }): string {
      // JSON.stringify 기반 안전 직렬화. sensitive 는 true 일 때만 포함.
      const parts: string[] = []
      parts.push(`targetId: ${JSON.stringify(target.targetId)}`)
      parts.push(`selector: ${JSON.stringify(target.selector)}`)
      // actionKinds 는 Plan 02 capture 에서 추론 불가 — 기본값 ['click'] 부여. 사용자가 수동 편집.
      parts.push(`actionKinds: ['click']`)
      if (target.sensitive) parts.push(`sensitive: true`)
      return `defineTarget({ ${parts.join(', ')} })`
    }
    ```

    **주의:** `actionKinds` 는 capture 시점에 알 수 없으므로 `['click']` 기본값. Sensitive 입력 필드인 경우 capture 단계에서 `isSensitive=true` 가 되지만 actionKind 는 여전히 click — 사용자가 commit 후 manifest.ts 에서 `'fill'` 로 수동 편집. 이는 의도된 동작 (recorder 는 100% 자동을 약속하지 않음).

    **NOTE on `diff` package:** `diff` 는 npm 에 있고 일부 transitive dep 로 이미 들어있을 가능성. 없으면 `pnpm --filter @agrune/mcp add diff @types/diff`.

    import { join, extname, resolve } from 'node:path'
    import { createInterface } from 'node:readline/promises'
    import { PendingStore, type PendingCaptureFile } from './pending-store.js'
    import { mergeTargetIntoManifest, MergeError } from './manifest-merger.js'

    const MAX_PENDING_SIZE = 256 * 1024  // 256 KB (T-16-15)

    export interface ManifestDevWatcherDeps {
      watcherFactory?: (root: string) => FSWatcher
      confirmPrompt?: (diff: string) => Promise<boolean>
      pendingStore?: PendingStore
      log?: { info: (m: string) => void; warn: (m: string) => void; error: (m: string) => void }
    }

    export class ManifestDevWatcher {
      private watcher: FSWatcher | null = null
      private readonly pendingStore: PendingStore
      private readonly confirmPrompt: (diff: string) => Promise<boolean>
      private readonly log: NonNullable<ManifestDevWatcherDeps['log']>
      private readonly watcherFactory: (root: string) => FSWatcher

      constructor(private readonly manifestPath: string, deps: ManifestDevWatcherDeps = {}) {
        this.pendingStore = deps.pendingStore ?? new PendingStore()
        this.confirmPrompt = deps.confirmPrompt ?? defaultConfirmPrompt
        this.log = deps.log ?? { info: console.error, warn: console.error, error: console.error }
        this.watcherFactory = deps.watcherFactory ?? ((root) => chokidar.watch(root, {
          persistent: true,
          ignoreInitial: false,
          depth: 2,
          awaitWriteFinish: { stabilityThreshold: 100, pollInterval: 50 },  // T-16-16
        }))
      }

      async start(): Promise<void> {
        const root = join(homedir(), '.agrune', 'authoring', 'pending')
        this.watcher = this.watcherFactory(root)
        this.watcher.on('add', (filePath) => {
          void this.processPending(filePath).catch((err) => {
            this.log.error(`[manifest dev] process error: ${err instanceof Error ? err.message : String(err)}`)
          })
        })
        this.log.info(`[manifest dev] watching ${root} → ${this.manifestPath}`)
      }

      async stop(): Promise<void> {
        if (this.watcher) { await this.watcher.close(); this.watcher = null }
      }

      private async processPending(filePath: string): Promise<void> {
        const st = await stat(filePath)
        if (st.size > MAX_PENDING_SIZE) {
          this.log.warn(`[manifest dev] skip large pending (>${MAX_PENDING_SIZE}B): ${filePath}`)
          return
        }
        const raw = await readFile(filePath, 'utf-8')
        let pending: PendingCaptureFile
        try { pending = JSON.parse(raw) as PendingCaptureFile }
        catch { this.log.error(`[manifest dev] invalid JSON: ${filePath}`); return }

        const sourceText = await readFile(this.manifestPath, 'utf-8')
        let result
        try {
          result = mergeTargetIntoManifest(sourceText, pending, this.manifestPath)
        } catch (err) {
          if (err instanceof MergeError) {
            this.log.warn(`[manifest dev] skip (${err.code}): ${err.message}`)
            return  // pending 유지 — 사용자가 JSON 수정 후 재감지
          }
          throw err
        }

        // Preview
        this.log.info(`--- pending: ${filePath} ---`)
        this.log.info(result.diff)
        const approved = await this.confirmPrompt(result.diff)
        if (!approved) {
          this.log.info(`[manifest dev] skipped (user declined)`)
          return
        }
        await writeFile(this.manifestPath, result.merged, 'utf-8')
        await this.pendingStore.deletePending(filePath)
        this.log.info(`[manifest dev] merged: ${result.addedTargetIds.join(', ')}`)
      }
    }

    async function defaultConfirmPrompt(_diff: string): Promise<boolean> {
      const rl = createInterface({ input: process.stdin, output: process.stderr })
      try {
        const ans = await rl.question('Apply? [y/N] ')
        return ans.trim().toLowerCase() === 'y'  // T-16-13: 정확 'y' 만 허용
      } finally { rl.close() }
    }

    export async function runManifestDevCli(args: string[]): Promise<number> {
      const manifestArg = args[0]
      if (!manifestArg) {
        process.stderr.write('Usage: agrune manifest dev <manifest.ts>\n')
        return 1
      }
      const abs = resolve(process.cwd(), manifestArg)
      const ext = extname(abs)
      if (ext !== '.ts' && ext !== '.tsx') {
        process.stderr.write(`[manifest dev] file must be .ts or .tsx: ${manifestArg}\n`)
        return 1
      }
      try { await stat(abs) } catch {
        process.stderr.write(`[manifest dev] file not found: ${abs}\n`)
        return 1
      }
      const watcher = new ManifestDevWatcher(abs)
      await watcher.start()
      // 프로세스 keep-alive — SIGINT 까지 지속
      await new Promise<void>((resolve) => {
        process.once('SIGINT', () => { void watcher.stop().finally(() => resolve()) })
        process.once('SIGTERM', () => { void watcher.stop().finally(() => resolve()) })
      })
      return 0
    }
