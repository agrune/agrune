    import { mkdir, writeFile, readdir, stat, rm } from 'node:fs/promises'
    import { join, relative, isAbsolute } from 'node:path'
    import { homedir } from 'node:os'

    const SESSION_ID_RE = /^[a-zA-Z0-9_-]{1,128}$/
    const TARGET_ID_RE = /^[a-zA-Z0-9_-]{1,64}$/
    const SEVEN_DAYS_MS = 7 * 24 * 60 * 60 * 1000

    export interface PendingCaptureFile { /* see interfaces */ }

    export class PendingStore {
      constructor(private readonly rootDir: string = join(homedir(), '.agrune', 'authoring', 'pending')) {}

      static sanitizeSessionId(id: string): string {
        if (!SESSION_ID_RE.test(id)) throw new Error(`Invalid sessionId: ${id}`)
        return id
      }
      static sanitizeTargetId(id: string): string {
        if (!TARGET_ID_RE.test(id)) throw new Error(`Invalid targetId: ${id}`)
        return id
      }

      async writePending(sessionId: string, payload: PendingCaptureFile): Promise<string> {
        const safe = PendingStore.sanitizeSessionId(sessionId)
        const dir = join(this.rootDir, safe)
        // Re-check containment (defense-in-depth, T-16-02)
        const rel = relative(this.rootDir, dir)
        if (rel.startsWith('..') || isAbsolute(rel)) throw new Error('Path traversal detected')
        await mkdir(dir, { recursive: true })
        const file = join(dir, `${payload.ts}.json`)
        await writeFile(file, JSON.stringify(payload, null, 2), 'utf-8')
        return file
      }

      async deletePending(filePath: string): Promise<void> {
        const rel = relative(this.rootDir, filePath)
        if (rel.startsWith('..') || isAbsolute(rel)) throw new Error('Path outside pending dir')
        await rm(filePath, { force: true })
      }

      async cleanup(maxAgeMs = SEVEN_DAYS_MS): Promise<number> { /* readdir + stat + rm for old files */ }
    }

    import type { SelectorLadder, FiberIdentityPath } from '@agrune/manifest'

    export interface CaptureResult { /* match devtools types.ts */ }

    export function buildRoleSelector(el: HTMLElement): { role: string; name?: string } {
      const role = el.getAttribute('role') ?? el.tagName.toLowerCase()
      const ariaLabel = el.getAttribute('aria-label')
      const text = el.textContent?.trim().slice(0, 80)
      const name = ariaLabel || text || undefined
      return name ? { role, name } : { role }
    }

    export function buildCssFallback(el: HTMLElement): string {
      const testId = el.getAttribute('data-testid')
      if (testId) return `[data-testid="${CSS.escape(testId)}"]`
      if (el.id && !/[a-zA-Z0-9]{8,}/.test(el.id)) return `#${CSS.escape(el.id)}`
      // path 기반 fallback — 해시 클래스 필터 (HASH_CLASS_PATTERN 과 동일 기준)
      return buildDomPath(el)
    }

    function buildDomPath(el: HTMLElement): string { /* nth-of-type 없이, stable attr 만 */ }

    export function captureElement(el: HTMLElement, counter: number): CaptureResult {
      const bridge = (window as Window & { __agrune_identity__?: { resolvePath?: (el: HTMLElement) => FiberIdentityPath | null; version?: string } }).__agrune_identity__
      let fiberPath: FiberIdentityPath | undefined
      if (bridge && typeof bridge.resolvePath === 'function') {
        fiberPath = bridge.resolvePath(el) ?? undefined
      }
      const role = buildRoleSelector(el)
      const css = buildCssFallback(el)
      const sensitive = isSensitive(el) ? true : undefined
      const autoTargetId = generateAutoTargetId(el, fiberPath, counter)
      return {
        url: window.location.href,
        ...(fiberPath ? { fiberPath } : {}),
        roleSelector: role,
        cssSelector: css,
        ...(sensitive ? { sensitive: true } : {}),
        autoTargetId,
      }
    }

    export function generateAutoTargetId(el: HTMLElement, path: FiberIdentityPath | undefined, counter: number): string {
      // Decision A: fiber componentName 기반. fallback 은 tagName.
      const last = path && path.length ? path[path.length - 1] : undefined
      const base = (last?.componentName || el.tagName.toLowerCase())
        .replace(/[^a-zA-Z0-9]/g, '')
        .slice(0, 48)
      return `${base || 'target'}_${counter}`
    }

    export function buildSelectorLadder(capture: CaptureResult): SelectorLadder {
      const ladder: Partial<SelectorLadder> = {}
      if (capture.fiberPath) ladder.fiber = { path: capture.fiberPath }
      if (capture.roleSelector?.role) ladder.role = capture.roleSelector.role
      if (capture.cssSelector) ladder.css = capture.cssSelector
      return ladder as SelectorLadder  // AtLeastOne 보장: css 가 항상 들어감
    }

    // Overlay: element hover 시 outline, click 시 captureElement 호출 → result 를 CDP 로 반환
    export function activateRecorderOverlay(onCapture: (r: CaptureResult) => void): () => void {
      let counter = 0
      let hovered: HTMLElement | null = null
      const setOutline = (el: HTMLElement | null, on: boolean) => {
        if (!el) return
        el.style.outline = on ? '2px solid #ff5722' : ''
      }
      const onMove = (e: MouseEvent) => {
        const t = e.target
        if (!(t instanceof HTMLElement) || t === hovered) return
        setOutline(hovered, false)
        hovered = t
        setOutline(t, true)
      }
      const onClick = (e: MouseEvent) => {
        e.preventDefault(); e.stopPropagation()
        const t = e.target
        if (!(t instanceof HTMLElement)) return
        setOutline(t, false)
        counter += 1
        const result = captureElement(t, counter)
        onCapture(result)
        cleanup()
      }
      const cleanup = () => {
        setOutline(hovered, false)
        document.removeEventListener('mousemove', onMove, true)
        document.removeEventListener('click', onClick, true)
      }
      document.addEventListener('mousemove', onMove, true)
      document.addEventListener('click', onClick, true)
      return cleanup
    }
